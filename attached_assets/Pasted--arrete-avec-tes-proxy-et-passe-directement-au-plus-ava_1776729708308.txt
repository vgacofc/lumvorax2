def build_krylov_basis(estimator, circuit, hamiltonian, m=6):
    states = []

    psi = circuit
    for k in range(m):
        exp_val = estimator.run([(psi, hamiltonian)]).result()[0]
        states.append(exp_val)

        # H|ψ>
        psi = psi.compose(circuit)  # approximation operatorielle

    return states

def build_hessenberg(krylov_states):
    import numpy as np

    m = len(krylov_states)
    Hm = np.zeros((m, m), dtype=complex)

    for i in range(m):
        for j in range(m):
            Hm[i, j] = np.vdot(krylov_states[i], krylov_states[j])

    return Hm

def pseudospectrum(Hm, eps=1e-2):
    import numpy as np

    eigvals = np.linalg.eigvals(Hm)

    return {
        "eigenvalues": eigvals.tolist(),
        "radius_instability": float(np.max(np.abs(eigvals))),
        "eps": eps
    }

def streaming_pseudospectrum(estimator, circuit, H, steps=6):
    krylov_vecs = []

    psi = circuit

    for k in range(steps):
        exp = estimator.run([(psi, H)]).result()[0]

        psi = psi.compose(circuit)

        yield {

            "instability": max(abs(v) for v in krylov_vecs)

def adaptive_krylov(H, psi0, noise_profile, m_max=10):

class SpectrumStream:
    def __init__(self):
        self.clients = set()

    async def handler(self, ws):
        self.clients.add(ws)
        try:
            async for msg in ws:
                pass
        finally:
            self.clients.remove(ws)

    async def broadcast(self, data):
        if self.clients:
            await asyncio.gather(*[
                c.send(json.dumps(data)) for c in self.clients
