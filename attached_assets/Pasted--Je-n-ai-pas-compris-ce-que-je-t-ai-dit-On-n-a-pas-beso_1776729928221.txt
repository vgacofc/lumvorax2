from fastapi import FastAPI
from qiskit_ibm_runtime import QiskitRuntimeService, Session, SamplerV2
import json

app = FastAPI()

@app.post("/run_circuit")
def run_circuit(payload: dict):
    service = QiskitRuntimeService(channel="ibm_quantum_platform")

    backend_name = payload.get("backend", "ibm_fez")
    qc = payload["circuit"]

    backend = service.backend(backend_name)

    with Session(service=service, backend=backend) as session:
        sampler = SamplerV2(session=session)
        job = sampler.run([qc], shots=payload.get("shots", 4096))
        result = job.result()

    return {
        "counts": result[0].data,
        "backend": backend_name
    }

from qiskit_ibm_runtime import QiskitRuntimeService, Session, SamplerV2

class IBMRuntimeManager:

    def __init__(self, backend_name="ibm_fez"):
        self.service = QiskitRuntimeService(channel="ibm_quantum_platform")
        self.backend = self.service.backend(backend_name)
        self.session = None
        self.sampler = None

    def start(self):
        self.session = Session(service=self.service, backend=self.backend)
        self.sampler = SamplerV2(session=self.session)

    def run_batch(self, circuits, shots=4096):
        return self.sampler.run(circuits, shots=shots).result()

    def close(self):
        if self.session:
            self.session.close()

2. MODULE 1 — Backend Router intelligent
Objectif
Choisir automatiquement :

Logique de sélection
Critères utilisés :
queue depth backend

Code (router intelligent)
from qiskit_ibm_runtime import QiskitRuntimeService

class IBMBackendRouter:

    def __init__(self):
        self.service = QiskitRuntimeService(channel="ibm_quantum_platform")

    def pick_backend(self, min_qubits=10, prefer_hw=True):

        backends = self.service.backends(operational=True)

        hw = []
        sim = []

        for b in backends:
            try:
                if b.num_qubits >= min_qubits:
                    if b.simulator:
                        sim.append(b)
                    else:
                        hw.append(b)
            except:
                continue

        # PRIORITY 1: hardware low queue
        if prefer_hw and hw:
            hw_sorted = sorted(hw, key=lambda b: b.status().pending_jobs)
            return hw_sorted[0]

        # PRIORITY 2: simulator runtime (low latency)
        if sim:
            return sim[0]

        # fallback
        return self.service.backend("aer_simulator")

4. MODULE 3 — Batching Runtime optimisé
Objectif
Maximiser throughput Qiskit Runtime

Pourquoi c’est critique
Sans batching :

1 circuit = 1 job API

Avec batching :

N circuits = 1 job

👉 réduction latence énorme

Code batch executor
from qiskit_ibm_runtime import SamplerV2, Session

class RuntimeBatchExecutor:

    def __init__(self, service, backend):
        self.session = Session(service=service, backend=backend)
        self.sampler = SamplerV2(session=self.session)

    def run_batch(self, circuits, shots=4096):

        job = self.sampler.run(circuits, shots=shots)
        result = job.result()

        return [
            pub.data for pub in result
        ]

    def close(self):
        self.session.close()
Optimisation clé
circuits groupés par type

mêmes layouts = même batch

réduction overhead IBM API

def fallback_execute(circuit):
