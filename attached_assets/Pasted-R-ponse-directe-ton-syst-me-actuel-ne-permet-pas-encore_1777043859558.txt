## 2.2 Implémentation (core)

```c
// correlation_extractor.c
#include <math.h>
#include <stdio.h>
#include <time.h>

#define EPS 1e-6

static uint64_t hash_u64(uint64_t x) {
    x ^= x >> 33;
    x *= 0xff51afd7ed558ccdULL;
    x ^= x >> 33;
    x *= 0xc4ceb9fe1a85ec53ULL;
    x ^= x >> 33;
    return x;
}

uint64_t compute_checksum(correlation_vector_t* cv) {
    uint64_t* data = (uint64_t*)cv;
    uint64_t h = 1469598103934665603ULL;

    for (size_t i = 0; i < sizeof(*cv)/8; i++) {
        h ^= hash_u64(data[i]);
    }
    return h;
}

void compute_gradients(correlation_vector_t* cv,
                       double (*E)(double,double)) {

    double th = cv->theta_h;
    double tu = cv->theta_u;

    cv->grad_theta_h =
        (E(th+EPS, tu) - E(th-EPS, tu)) / (2*EPS);

    cv->grad_theta_u =
        (E(th, tu+EPS) - E(th, tu-EPS)) / (2*EPS);
}

void compute_curvature(correlation_vector_t* cv,
                       double (*E)(double,double)) {

    double th = cv->theta_h;
    double tu = cv->theta_u;

    cv->curv_theta_h =
        (E(th+EPS, tu) - 2*E(th,tu) + E(th-EPS,tu)) / (EPS*EPS);

    cv->curv_theta_u =
        (E(th, tu+EPS) - 2*E(th,tu) + E(th,tu-EPS)) / (EPS*EPS);
}

void log_correlation_vector(const correlation_vector_t* cv,
                            const char* path) {

    FILE* f = fopen(path, "a");

    fprintf(f,
        "%lu,%lu,%.8f,%.8f,%.8f,%.8f,%.8f,%.8f,%.8f,%.8f,%lu\n",
        cv->iter,
        cv->timestamp_ns,
        cv->theta_h,
        cv->theta_u,
        cv->energy,
        cv->S_pi,
        cv->grad_theta_h,
        cv->grad_theta_u,
        cv->curv_theta_h,
        cv->curv_theta_u,
        cv->checksum_state
    );

    fclose(f);
}
```
