```c
// PATTERN OBLIGATOIRE TOUTES FONCTIONS CRITIQUES
bool secure_function_template(const void* input, size_t input_size, void** output) {
    // 1. Validation paramètres obligatoire
    if (!input || input_size == 0 || !output) return false;

    // 2. Vérification limites sécuritaires
    if (input_size > MAX_SECURE_BUFFER_SIZE) return false;

    // 3. Allocation trackée obligatoire  
    *output = TRACKED_MALLOC(input_size);
    if (!*output) return false;

    // 4. Opération avec validation continue
    bool success = perform_operation(input, input_size, *output);

    // 5. Nettoyage sécurisé obligatoire si échec
    if (!success) {
        secure_zero_memory(*output, input_size);
        TRACKED_FREE(*output);
        *output = NULL;
    }

    // 6. Logging forensique systématique
    forensic_log(success ? FORENSIC_LEVEL_INFO : FORENSIC_LEVEL_ERROR,
                "secure_function", "Operation: %s", success ? "SUCCESS" : "FAILURE");

    return success;
}
```
